<script>

  // Hier tragen Sie die Dateinamen Ihrer MP3s aus dem "music"-Ordner ein:

  const songs = [

    { title: "Artlist Track 1", file: "music/song1.mp3" },

    { title: "Artlist Track 2", file: "music/song2.mp3" },

    { title: "Artlist Track 3", file: "music/song3.mp3" }

  ];

  let currentIndex = 0;

  const audioPlayer = document.getElementById('audioPlayer');

  const trackTitle = document.getElementById('currentTrackTitle');

  const playlistElement = document.getElementById('playlist');

  function renderPlaylist() {

    playlistElement.innerHTML = '';

    songs.forEach((song, index) => {

      const li = document.createElement('li');

      li.textContent = `${index + 1}. ${song.title}`;

      if (index === currentIndex) li.classList.add('active');

      li.onclick = () => loadAndPlay(index);

      playlistElement.appendChild(li);

    });

  }

  function loadAndPlay(index) {

    currentIndex = index;

    audioPlayer.src = songs[currentIndex].file;

    trackTitle.textContent = songs[currentIndex].title;

    renderPlaylist();

    audioPlayer.play().catch(() => {

      // In manchen Browsern verhindert Autoplay den ersten Start ohne Klick

    });

  }

  // Automatisch den nächsten Song abspielen, wenn einer zu Ende ist

  audioPlayer.onended = () => {

    currentIndex = (currentIndex + 1) % songs.length;

    loadAndPlay(currentIndex);

  };

  // Ersten Song beim Laden bereitstellen

  loadAndPlay(0);

</script>
